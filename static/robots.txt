# sitemap lists every one of them, so nothing needs a query string to be found.

Allow: /

# Endpoints, not pages.
Disallow: /api/
Disallow: /auth/

# Yours alone, and nothing a search result could usefully point at.
Disallow: /account

# different query. Blocking them keeps a crawler off the Mongo-backed routes,
# which matters more here than usual: the cluster's read budget is small, and
# crawling /players?sort=…&page=… would spend it on duplicates.
Disallow: /*?
